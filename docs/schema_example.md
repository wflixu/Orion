# Orion Schema DSL Example

## 概述

Schema DSL 提供声明式表定义，生成类型安全的 MoonBit 代码。

## 定义 Schema

```moonbit
let user_schema = @schema.schema("users")
  |> @schema.add_int_field("id", @schema.pk_auto_inc())
  |> @schema.add_string_field("name", @schema.required())
  |> @schema.add_string_field("email", @schema.unique_required())
  |> @schema.add_int_field("age", [])
  |> @schema.add_bool_field("active", @schema.with_default("true"))
  |> @schema.add_normal_index("idx_name", ["name"])
  |> @schema.add_unique_index("uniq_email", ["email"])
  |> @schema.build_schema()
```

## 生成代码

```moonbit
// 1. 生成 Struct
let struct_code = @codegen.gen_full_struct(user_schema)

// 2. 生成 CRUD Mapper
let crud_code = @codegen.gen_full_crud(user_schema)

// 3. 生成 Query Builder
let query_code = @codegen.gen_full_query_builder(user_schema)

// 4. 生成完整模块
let full_code = @codegen.gen_full_struct(user_schema) +
                @codegen.gen_full_crud(user_schema) +
                @codegen.gen_full_query_builder(user_schema)
```

### CRUD Mapper
```moonbit
// @generated by Orion Codegen 2.0
// 请勿手动编辑此文件

pub struct UserMapper {
  db : @runtime.Db
}

pub fn new_user_mapper(db : @runtime.Db) -> UserMapper {
  UserMapper { db }
}

pub fn create_user(self : UserMapper, name : String, email : String, age : Int, active : Bool) -> Result[Int, @runtime.DbError] {
  let sql = "INSERT INTO users (name, email, age, active) VALUES (?, ?, ?, ?)"
  @runtime.execute(self.db, sql, [@runtime.DbValue::String(name), @runtime.DbValue::String(email), @runtime.DbValue::Int(age), @runtime.DbValue::Bool(active)])
}

pub fn find_user_by_id(self : UserMapper, id : Int) -> Result[Option[User], @runtime.DbError] {
  let sql = "SELECT id, name, email, age, active FROM users WHERE id = ?"
  match @runtime.query(self.db, sql, [@runtime.DbValue::Int(id)]) {
    Ok(rows) => if rows.length() > 0 { Ok(Some(User::from_row(rows[0]))) } else { Ok(None) }
    Err(e) => Err(e)
  }
}

pub fn find_all_users(self : UserMapper) -> Result[List[User], @runtime.DbError] {
  let sql = "SELECT id, name, email, age, active FROM users"
  match @runtime.query(self.db, sql, []) {
    Ok(rows) => {
      let mut result : List[User] = []
      for row in rows {
        result.push(User::from_row(row))
      }
      Ok(result)
    }
    Err(e) => Err(e)
  }
}

pub fn update_user(self : UserMapper, id : Int, name : String, email : String, age : Int, active : Bool) -> Result[Int, @runtime.DbError] {
  let sql = "UPDATE users SET name = ?, email = ?, age = ?, active = ? WHERE id = ?"
  @runtime.execute(self.db, sql, [@runtime.DbValue::String(name), @runtime.DbValue::String(email), @runtime.DbValue::Int(age), @runtime.DbValue::Bool(active), @runtime.DbValue::Int(id)])
}

pub fn delete_user_by_id(self : UserMapper, id : Int) -> Result[Int, @runtime.DbError] {
  let sql = "DELETE FROM users WHERE id = ?"
  @runtime.execute(self.db, sql, [@runtime.DbValue::Int(id)])
}
```

### Query Builder
```moonbit
// @generated by Orion Codegen 2.0
// 请勿手动编辑此文件

pub struct UsersQuery {
  db : @runtime.Db
  where_clauses : Array[String]
  params : Array[@runtime.DbValue]
  order_by : Option[String]
  limit : Option[Int]
  offset : Option[Int]
}

pub fn new_users_query(db : @runtime.Db) -> UsersQuery {
  UsersQuery {
    db,
    where_clauses: [],
    params: [],
    order_by: None,
    limit: None,
    offset: None,
  }
}

pub fn UsersQuery::where_id(mut self, value : Int) -> Self {
  self.where_clauses.push("id = ?")
  self.params.push(@runtime.DbValue::Int(value))
  self
}

pub fn UsersQuery::where_name(mut self, value : String) -> Self {
  self.where_clauses.push("name = ?")
  self.params.push(@runtime.DbValue::String(value))
  self
}

pub fn UsersQuery::where_name_contains(mut self, pattern : String) -> Self {
  self.where_clauses.push("name LIKE ?")
  self.params.push(@runtime.DbValue::String("%" + pattern + "%"))
  self
}

pub fn UsersQuery::where_name_starts_with(mut self, prefix : String) -> Self {
  self.where_clauses.push("name LIKE ?")
  self.params.push(@runtime.DbValue::String(prefix + "%"))
  self
}

pub fn UsersQuery::where_name_ends_with(mut self, suffix : String) -> Self {
  self.where_clauses.push("name LIKE ?")
  self.params.push(@runtime.DbValue::String("%" + suffix))
  self
}

pub fn UsersQuery::order_by_id(mut self, desc : Bool = false) -> Self {
  self.order_by = Some("id " + (if desc { "DESC" } else { "ASC" }))
  self
}

pub fn UsersQuery::limit(mut self, n : Int) -> Self {
  self.limit = Some(n)
  self
}

pub fn UsersQuery::offset(mut self, n : Int) -> Self {
  self.offset = Some(n)
  self
}

pub fn UsersQuery::execute(self) -> Result[List[User], @runtime.DbError] {
  let mut sql = "SELECT id, name, email, age, active FROM users"

  if self.where_clauses.length() > 0 {
    sql = sql + " WHERE " + String::join(self.where_clauses, " AND ")
  }

  if self.order_by != None {
    sql = sql + " ORDER BY " + self.order_by.unwrap()
  }

  if self.limit != None {
    sql = sql + " LIMIT " + Int::to_string(self.limit.unwrap())
  }

  if self.offset != None {
    sql = sql + " OFFSET " + Int::to_string(self.offset.unwrap())
  }

  match @runtime.query(self.db, sql, self.params) {
    Ok(rows) => {
      let mut result : List[User] = []
      for row in rows {
        result.push(User::from_row(row))
      }
      Ok(result)
    }
    Err(e) => Err(e)
  }
}

pub fn UsersQuery::first(self) -> Result[Option[User], @runtime.DbError] {
  self.limit(1).execute().map(fn(list) => list.head())
}

pub fn UsersQuery::count(mut self) -> Result[Int, @runtime.DbError] {
  let mut sql = "SELECT COUNT(*) FROM users"

  if self.where_clauses.length() > 0 {
    sql = sql + " WHERE " + String::join(self.where_clauses, " AND ")
  }

  match @runtime.query(self.db, sql, self.params) {
    Ok(rows) => {
      if rows.length() > 0 {
        match rows[0] {
          @runtime.DbValue::Int(count) => Ok(count)
          _ => Ok(0)
        }
      } else {
        Ok(0)
      }
    }
    Err(e) => Err(e)
  }
}
```

## 使用示例

```moonbit
// 创建 Mapper
let mapper = new_user_mapper(db)

// 创建记录
let id = create_user(mapper, "Alice", "alice@example.com", 30, true)

// 查询单个记录
let user = find_user_by_id(mapper, id)

// 查询所有记录
let all_users = find_all_users(mapper)

// 使用 Query Builder
let active_users = new_users_query(db)
  |> .where_active(true)
  |> .order_by_name(false)
  |> .limit(10)
  |> .execute()

// 条件查询
let alice = new_users_query(db)
  |> .where_name_contains("ali")
  |> .first()

// 计数
let count = new_users_query(db)
  |> .where_active(true)
  |> .count()
```

## 约束验证

Schema DSL 在构建时自动验证约束：

```moonbit
// AutoInc 必须与 PrimaryKey 一起使用
let invalid = @schema.schema("test")
  |> @schema.add_int_field("id", @schema.auto_inc()) // 错误！
  |> @schema.build_schema()

// 正确的做法
let valid = @schema.schema("test")
  |> @schema.add_int_field("id", @schema.pk_auto_inc()) // 正确
  |> @schema.build_schema()
```
